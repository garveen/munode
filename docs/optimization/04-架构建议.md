# MuNode 架构建议

## 1. 共享配置 Schema

使用 Zod 进行运行时验证，确保配置的类型安全和验证。

### 1.1 配置 Schema 定义

```typescript
// common/src/config/schemas.ts
import { z } from 'zod';

export const NetworkConfigSchema = z.object({
  host: z.string().ip().default('0.0.0.0'),
  port: z.number().min(1).max(65535),
  externalHost: z.string().optional(),
});

export const TLSConfigSchema = z.object({
  enabled: z.boolean().default(true),
  certFile: z.string(),
  keyFile: z.string(),
  caFile: z.string().optional(),
  rejectUnauthorized: z.boolean().default(true),
});

export const LogConfigSchema = z.object({
  level: z.enum(['error', 'warn', 'info', 'debug']).default('info'),
  file: z.string().optional(),
  format: z.enum(['json', 'text']).default('text'),
});

export const DatabaseConfigSchema = z.object({
  path: z.string(),
  backupDir: z.string().optional(),
  backupInterval: z.number().optional(),
});

export const AuthConfigSchema = z.object({
  allowAnonymous: z.boolean().default(false),
  requireCertificate: z.boolean().default(false),
  externalAuthUrl: z.string().url().optional(),
  hmacSecret: z.string().optional(),
});

// Hub 配置 Schema
export const HubConfigSchema = z.object({
  host: z.string().default('0.0.0.0'),
  controlPort: z.number().min(1).max(65535),
  database: DatabaseConfigSchema,
  tls: TLSConfigSchema,
  log: LogConfigSchema.optional(),
  auth: AuthConfigSchema.optional(),
});

// Edge 配置 Schema
export const EdgeConfigSchema = z.object({
  server_id: z.number(),
  name: z.string(),
  network: NetworkConfigSchema,
  tls: TLSConfigSchema,
  hubServer: z.object({
    host: z.string(),
    controlPort: z.number(),
    hmacSecret: z.string().optional(),
  }),
  log: LogConfigSchema.optional(),
});

// 类型提取
export type HubConfig = z.infer<typeof HubConfigSchema>;
export type EdgeConfig = z.infer<typeof EdgeConfigSchema>;

// 验证并加载配置
export function loadConfig<T extends z.ZodType>(
  schema: T,
  configPath: string
): z.infer<T> {
  const rawConfig = JSON.parse(fs.readFileSync(configPath, 'utf-8'));
  return schema.parse(rawConfig);
}

// 使用示例
const hubConfig = loadConfig(HubConfigSchema, './config/hub.json');
const edgeConfig = loadConfig(EdgeConfigSchema, './config/edge.json');
```

### 1.2 优点

- **类型安全**：编译时和运行时都有类型检查
- **自动验证**：加载配置时自动验证格式和值
- **清晰的错误信息**：Zod 提供详细的验证错误
- **默认值支持**：可以定义可选字段的默认值
- **文档即代码**：Schema 本身就是配置文档

```typescript
export function createSecureTLSOptions(config: {
  certFile: string;
  keyFile: string;
  caFile?: string;
}): TLSSocketOptions {
  return {
    // 最低 TLS 版本

    ca: config.caFile ? fs.readFileSync(config.caFile) : undefined,
    
    // 请求客户端证书（可选）
    requestCert: false,
    rejectUnauthorized: true,
  };
}
```

```typescript
// 证书有效期检查
export function checkCertificateExpiry(certPath: string): {
  valid: boolean;
  daysUntilExpiry: number;
  expiryDate: Date;
} {
  const cert = fs.readFileSync(certPath);
  const x509 = new crypto.X509Certificate(cert);
  
  const expiryDate = new Date(x509.validTo);
  const now = new Date();
  const daysUntilExpiry = Math.floor(
    (expiryDate.getTime() - now.getTime()) / (1000 * 60 * 60 * 24)
  );
  
  return {
    valid: daysUntilExpiry > 0,
    daysUntilExpiry,
    expiryDate,
  };
}

// 定期检查证书
export class CertificateMonitor {
  private checkInterval: NodeJS.Timeout;
  
  constructor(
    private certPath: string,
    private warningDays: number = 30,
    private logger: Logger
  ) {}
  
  start(): void {
    // 每天检查一次
    this.checkInterval = setInterval(() => {
      const { valid, daysUntilExpiry, expiryDate } = checkCertificateExpiry(this.certPath);
      
      if (!valid) {
        this.logger.error(`Certificate has expired! Expiry: ${expiryDate.toISOString()}`);
      } else if (daysUntilExpiry <= this.warningDays) {
        this.logger.warn(
          `Certificate expires in ${daysUntilExpiry} days (${expiryDate.toISOString()})`
        );
      }
    }, 24 * 60 * 60 * 1000);
  }
  
  stop(): void {
    if (this.checkInterval) {
      clearInterval(this.checkInterval);
    }
  }
}
```

## 3. 速率限制

### 3.1 连接速率限制

```typescript
// common/src/security/rate-limiter.ts

export interface RateLimitConfig {
  windowMs: number;      // 时间窗口（毫秒）
  maxRequests: number;   // 窗口内最大请求数
  blockDurationMs?: number; // 超限后封锁时间
}

export class ConnectionRateLimiter {
  private connections: Map<string, number[]> = new Map();
  private blocked: Map<string, number> = new Map();
  
  constructor(private config: RateLimitConfig) {}
  
  canConnect(ip: string): { allowed: boolean; retryAfterMs?: number } {
    const now = Date.now();
    
    // 检查是否被封锁
    const blockedUntil = this.blocked.get(ip);
    if (blockedUntil && now < blockedUntil) {
      return { 
        allowed: false, 
        retryAfterMs: blockedUntil - now 
      };
    }
    
    // 清理过期的封锁
    if (blockedUntil && now >= blockedUntil) {
      this.blocked.delete(ip);
    }
    
    // 获取连接历史
    const history = this.connections.get(ip) || [];
    
    // 清理超出时间窗口的记录
    const windowStart = now - this.config.windowMs;
    const recent = history.filter(t => t >= windowStart);
    
    // 检查是否超限
    if (recent.length >= this.config.maxRequests) {
      // 设置封锁
      if (this.config.blockDurationMs) {
        this.blocked.set(ip, now + this.config.blockDurationMs);
      }
      return { 
        allowed: false, 
        retryAfterMs: this.config.blockDurationMs || this.config.windowMs 
      };
    }
    
    // 记录本次连接
    recent.push(now);
    this.connections.set(ip, recent);
    
    return { allowed: true };
  }
  
  // 手动封锁 IP
  blockIP(ip: string, durationMs: number): void {
    this.blocked.set(ip, Date.now() + durationMs);
  }
  
  // 解除封锁
  unblockIP(ip: string): void {
    this.blocked.delete(ip);
    this.connections.delete(ip);
  }
  
  // 获取当前状态
  getStatus(ip: string): {
    requestsInWindow: number;
    isBlocked: boolean;
    blockedUntil?: Date;
  } {
    const now = Date.now();
    const history = this.connections.get(ip) || [];
    const windowStart = now - this.config.windowMs;
    const recent = history.filter(t => t >= windowStart);
    
    const blockedUntil = this.blocked.get(ip);
    
    return {
      requestsInWindow: recent.length,
      isBlocked: blockedUntil ? now < blockedUntil : false,
      blockedUntil: blockedUntil ? new Date(blockedUntil) : undefined,
    };
  }
  
  // 定期清理过期数据
  cleanup(): void {
    const now = Date.now();
    const windowStart = now - this.config.windowMs;
    
    for (const [ip, history] of this.connections) {
      const recent = history.filter(t => t >= windowStart);
      if (recent.length === 0) {
        this.connections.delete(ip);
      } else {
        this.connections.set(ip, recent);
      }
    }
    
    for (const [ip, blockedUntil] of this.blocked) {
      if (now >= blockedUntil) {
        this.blocked.delete(ip);
      }
    }
  }
}

// 使用示例
const rateLimiter = new ConnectionRateLimiter({
  windowMs: 60000,         // 1分钟窗口
  maxRequests: 10,         // 每分钟最多10次连接
  blockDurationMs: 300000, // 超限后封锁5分钟
});

// 在连接处理中使用
server.on('connection', (socket) => {
  const ip = socket.remoteAddress;
  const { allowed, retryAfterMs } = rateLimiter.canConnect(ip);
  
  if (!allowed) {
    logger.warn(`Rate limited: ${ip}, retry after ${retryAfterMs}ms`);
    socket.destroy();
    return;
  }
  
  // 继续处理连接...
});
```

```typescript
// 每用户的消息速率限制
export class MessageRateLimiter {
  private userLimits: Map<number, { count: number; resetTime: number }> = new Map();
  
  constructor(
    private maxMessagesPerSecond: number = 5,
    private burstLimit: number = 10
  ) {}
  
  canSendMessage(sessionId: number): boolean {
    const now = Date.now();
    const limit = this.userLimits.get(sessionId);
    
    if (!limit || now >= limit.resetTime) {
      // 新的时间窗口
      this.userLimits.set(sessionId, {
        count: 1,
        resetTime: now + 1000,
      });
      return true;
    }
    
    // 检查是否超限
    if (limit.count >= this.burstLimit) {
      return false;
    }
    
    limit.count++;
    return true;
  }
  
  reset(sessionId: number): void {
    this.userLimits.delete(sessionId);
  }
}
```

## 4. 认证强化

### 4.1 HMAC 认证增强

```typescript
import crypto from 'crypto';

// HMAC 认证工具
export class HMACAuth {
  constructor(private secret: string) {}
  
  // 生成带时间戳的 challenge
  generateChallenge(): string {
    const timestamp = Date.now().toString();
    const nonce = crypto.randomBytes(16).toString('hex');
    return `${timestamp}:${nonce}`;
  }
  
  // 计算 HMAC 响应
  computeResponse(challenge: string): string {
    return crypto
      .createHmac('sha256', this.secret)
      .update(challenge)
      .digest('hex');
  }
  
  // 验证 HMAC 响应
  validateResponse(
    challenge: string,
    response: string,
    maxAgeMs: number = 60000
  ): { valid: boolean; error?: string } {
    // 解析 challenge
    const parts = challenge.split(':');
    if (parts.length !== 2) {
      return { valid: false, error: 'Invalid challenge format' };
    }
    
    const [timestampStr] = parts;
    const timestamp = parseInt(timestampStr, 10);
    
    if (isNaN(timestamp)) {
      return { valid: false, error: 'Invalid timestamp' };
    }
    
    // 检查时间窗口
    const age = Date.now() - timestamp;
    if (age > maxAgeMs) {
      return { valid: false, error: 'Challenge expired' };
    }
    
    if (age < 0) {
      return { valid: false, error: 'Challenge from future' };
    }
    
    // 计算预期响应
    const expected = this.computeResponse(challenge);
    
    // 使用时间安全的比较
    try {
      const isValid = crypto.timingSafeEqual(
        Buffer.from(response, 'hex'),
        Buffer.from(expected, 'hex')
      );
      
      return { valid: isValid, error: isValid ? undefined : 'Invalid response' };
    } catch {
      return { valid: false, error: 'Invalid response format' };
    }
  }
}

// 使用示例
const auth = new HMACAuth('your-secret-key');

// Edge 注册时
const challenge = auth.generateChallenge();
// 发送 challenge 给 Edge

// Edge 响应
const response = auth.computeResponse(challenge);

// Hub 验证
const result = auth.validateResponse(challenge, response);
if (!result.valid) {
  console.error('Authentication failed:', result.error);
}
```

```typescript
// 证书指纹验证
export function getCertificateFingerprint(cert: Buffer | string): string {
  const x509 = new crypto.X509Certificate(cert);
  return x509.fingerprint256;
}

// 证书信任链验证
export class CertificateTrustStore {
  private trustedFingerprints: Set<string> = new Set();
  
  addTrusted(cert: Buffer | string): void {
    const fingerprint = getCertificateFingerprint(cert);
    this.trustedFingerprints.add(fingerprint);
  }
  
  removeTrusted(fingerprint: string): void {
    this.trustedFingerprints.delete(fingerprint);
  }
  
  isTrusted(cert: Buffer | string): boolean {
    const fingerprint = getCertificateFingerprint(cert);
    return this.trustedFingerprints.has(fingerprint);
  }
  
  getAllTrusted(): string[] {
    return Array.from(this.trustedFingerprints);
  }
}

// 客户端证书验证
export function validateClientCertificate(
  cert: crypto.X509Certificate,
  options: {
    requireCN?: string;
    requireOU?: string;
    maxValidDays?: number;
  } = {}
): { valid: boolean; errors: string[] } {
  const errors: string[] = [];
  
  // 检查有效期
  const now = new Date();
  const validFrom = new Date(cert.validFrom);
  const validTo = new Date(cert.validTo);
  
  if (now < validFrom) {
    errors.push('Certificate not yet valid');
  }
  
  if (now > validTo) {
    errors.push('Certificate has expired');
  }
  
  if (options.maxValidDays) {
    const daysUntilExpiry = Math.floor(
      (validTo.getTime() - now.getTime()) / (1000 * 60 * 60 * 24)
    );
    if (daysUntilExpiry > options.maxValidDays) {
      errors.push(`Certificate valid for too long (${daysUntilExpiry} days)`);
    }
  }
  
  // 检查 CN
  if (options.requireCN) {
    const cn = cert.subject.split(',').find(s => s.trim().startsWith('CN='));
    if (!cn || !cn.includes(options.requireCN)) {
      errors.push(`CN mismatch: expected ${options.requireCN}`);
    }
  }
  
  // 检查 OU
  if (options.requireOU) {
    const ou = cert.subject.split(',').find(s => s.trim().startsWith('OU='));
    if (!ou || !ou.includes(options.requireOU)) {
      errors.push(`OU mismatch: expected ${options.requireOU}`);
    }
  }
  
  return {
    valid: errors.length === 0,
    errors,
  };
}
```

### 4.3 登录尝试限制

```typescript
// 登录尝试限制
export class LoginAttemptLimiter {
  private attempts: Map<string, { count: number; lockedUntil?: number }> = new Map();
  
  constructor(
    private maxAttempts: number = 5,
    private lockoutDurationMs: number = 900000 // 15分钟
  ) {}
  
  recordFailedAttempt(identifier: string): {
    locked: boolean;
    attemptsRemaining: number;
    lockedUntil?: Date;
  } {
    const now = Date.now();
    const record = this.attempts.get(identifier) || { count: 0 };
    
    // 检查是否已锁定
    if (record.lockedUntil && now < record.lockedUntil) {
      return {
        locked: true,
        attemptsRemaining: 0,
        lockedUntil: new Date(record.lockedUntil),
      };
    }
    
    // 如果锁定已过期，重置
    if (record.lockedUntil && now >= record.lockedUntil) {
      record.count = 0;
      record.lockedUntil = undefined;
    }
    
    // 增加失败计数
    record.count++;
    
    // 检查是否需要锁定
    if (record.count >= this.maxAttempts) {
      record.lockedUntil = now + this.lockoutDurationMs;
      this.attempts.set(identifier, record);
      return {
        locked: true,
        attemptsRemaining: 0,
        lockedUntil: new Date(record.lockedUntil),
      };
    }
    
    this.attempts.set(identifier, record);
    return {
      locked: false,
      attemptsRemaining: this.maxAttempts - record.count,
    };
  }
  
  recordSuccessfulLogin(identifier: string): void {
    this.attempts.delete(identifier);
  }
  
  isLocked(identifier: string): boolean {
    const record = this.attempts.get(identifier);
    if (!record || !record.lockedUntil) return false;
    return Date.now() < record.lockedUntil;
  }
  
  unlock(identifier: string): void {
    this.attempts.delete(identifier);
  }
}
```
