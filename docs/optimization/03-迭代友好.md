### 2.2 插件管理器

```typescript
// common/src/plugin/plugin-manager.ts

export class PluginManager<T extends EdgePlugin | HubPlugin> {
  private plugins: Map<string, T> = new Map();
  private context: PluginContext;
  
  constructor(context: PluginContext) {
    this.context = context;
  }
  
  async loadPlugin(pluginPath: string): Promise<void> {
    const plugin = await import(pluginPath);
    const instance: T = new plugin.default();
    
    // 验证元数据
    if (!instance.metadata?.name) {
      throw new Error('Plugin must have metadata.name');
    }
    
    // 检查依赖
    if (instance.metadata.dependencies) {
      for (const dep of instance.metadata.dependencies) {
        if (!this.plugins.has(dep)) {
          throw new Error(`Missing dependency: ${dep}`);
        }
      }
    }
    
    // 调用加载钩子
    if (instance.onLoad) {
      await instance.onLoad(this.context);
    }
    
    this.plugins.set(instance.metadata.name, instance);
    this.context.logger.info(`Plugin loaded: ${instance.metadata.name}`);
  }
  
  async unloadPlugin(name: string): Promise<void> {
    const plugin = this.plugins.get(name);
    if (!plugin) return;
    
    if (plugin.onUnload) {
      await plugin.onUnload();
    }
    
    this.plugins.delete(name);
    this.context.logger.info(`Plugin unloaded: ${name}`);
  }
  
  // 钩子调用
  callHook<K extends keyof T>(
    hookName: K,
    ...args: T[K] extends (...args: infer P) => any ? P : never
  ): void {
    for (const plugin of this.plugins.values()) {
      const hook = plugin[hookName];
      if (typeof hook === 'function') {
        try {
          hook.apply(plugin, args);
        } catch (error) {
          this.context.logger.error(`Plugin ${plugin.metadata.name} hook ${String(hookName)} error:`, error);
        }
      }
    }
  }
  
  // 链式钩子调用（可修改数据）
  callChainHook<K extends keyof T, R>(
    hookName: K,
    initialValue: R,
    ...args: any[]
  ): R | null {
    let value: R | null = initialValue;
    
    for (const plugin of this.plugins.values()) {
      if (value === null) break; // 被某个插件丢弃
      
      const hook = plugin[hookName];
      if (typeof hook === 'function') {
        try {
          value = hook.apply(plugin, [value, ...args]);
        } catch (error) {
          this.context.logger.error(`Plugin ${plugin.metadata.name} hook ${String(hookName)} error:`, error);
        }
      }
    }
    
    return value;
  }
}
```

### 3.1 指标收集器

```typescript
// common/src/metrics/metrics-collector.ts

export interface MetricValue {
  value: number;
  labels?: Record<string, string>;
  timestamp: number;
}

export class MetricsCollector {
  private metrics: Map<string, MetricValue[]> = new Map();
  private counters: Map<string, number> = new Map();
  private gauges: Map<string, number> = new Map();
  
  // 计数器
  increment(name: string, labels?: Record<string, string>, amount = 1): void {
    const key = this.makeKey(name, labels);
    const current = this.counters.get(key) || 0;
    this.counters.set(key, current + amount);
  }
  
  // 仪表
  setGauge(name: string, value: number, labels?: Record<string, string>): void {
    const key = this.makeKey(name, labels);
    this.gauges.set(key, value);
  }
  
  // 直方图/计时
  recordTiming(name: string, durationMs: number, labels?: Record<string, string>): void {
    const key = this.makeKey(name, labels);
    const values = this.metrics.get(key) || [];
    values.push({ value: durationMs, labels, timestamp: Date.now() });
    
    // 保留最近1000条
    if (values.length > 1000) {
      values.shift();
    }
    
    this.metrics.set(key, values);
  }
  
  // 获取统计
  getStats(name: string, labels?: Record<string, string>): {
    count: number;
    min: number;
    max: number;
    avg: number;
    p50: number;
    p95: number;
    p99: number;
  } | null {
    const key = this.makeKey(name, labels);
    const values = this.metrics.get(key);
    if (!values || values.length === 0) return null;
    
    const nums = values.map(v => v.value).sort((a, b) => a - b);
    const count = nums.length;
    
    return {
      count,
      min: nums[0],
      max: nums[count - 1],
      avg: nums.reduce((a, b) => a + b, 0) / count,
      p50: nums[Math.floor(count * 0.5)],
      p95: nums[Math.floor(count * 0.95)],
      p99: nums[Math.floor(count * 0.99)],
    };
  }
  
  // 导出 Prometheus 格式
  exportPrometheus(): string {
    const lines: string[] = [];
    
    for (const [key, value] of this.counters) {
      lines.push(`${key} ${value}`);
    }
    
    for (const [key, value] of this.gauges) {
      lines.push(`${key} ${value}`);
    }
    
    return lines.join('\n');
  }
  
  private makeKey(name: string, labels?: Record<string, string>): string {
    if (!labels || Object.keys(labels).length === 0) {
      return name;
    }
    const labelStr = Object.entries(labels)
      .map(([k, v]) => `${k}="${v}"`)
      .join(',');
    return `${name}{${labelStr}}`;
  }
}

// 全局单例
export const metrics = new MetricsCollector();
```

```typescript
// 健康检查接口
interface HealthStatus {
  status: 'healthy' | 'degraded' | 'unhealthy';
  checks: {
    [key: string]: {
      status: 'pass' | 'warn' | 'fail';
      message?: string;
      lastCheck: number;
    };
  };
  uptime: number;
}

export class HealthChecker {
  private checks: Map<string, () => Promise<boolean>> = new Map();
  
  register(name: string, check: () => Promise<boolean>): void {
    this.checks.set(name, check);
  }
  
  async getStatus(): Promise<HealthStatus> {
    const results: HealthStatus['checks'] = {};
    let hasFailure = false;
    let hasWarning = false;
    
    for (const [name, check] of this.checks) {
      try {
        const passed = await check();
        results[name] = {
          status: passed ? 'pass' : 'warn',
          lastCheck: Date.now(),
        };
        if (!passed) hasWarning = true;
      } catch (error) {
        results[name] = {
          status: 'fail',
          message: error.message,
          lastCheck: Date.now(),
        };
        hasFailure = true;
      }
    }
    
    return {
      status: hasFailure ? 'unhealthy' : hasWarning ? 'degraded' : 'healthy',
      checks: results,
      uptime: process.uptime(),
    };
  }
}
```

```typescript
// tests/helpers/integration-env.ts

export class IntegrationTestEnv {
  private hub: HubServer;
  private edges: EdgeServer[] = [];
  private clients: TestClient[] = [];
  
  async setup(options: {
    edgeCount?: number;
    clientsPerEdge?: number;
  } = {}): Promise<void> {
    const { edgeCount = 1, clientsPerEdge = 2 } = options;
    
    // 启动 Hub
    this.hub = await this.startHub();
    
    // 启动 Edges
    for (let i = 0; i < edgeCount; i++) {
      const edge = await this.startEdge(i);
      this.edges.push(edge);
    }
    
    // 创建测试客户端
    for (const edge of this.edges) {
      for (let i = 0; i < clientsPerEdge; i++) {
        const client = await this.createClient(edge);
        this.clients.push(client);
      }
    }
  }
  
  async teardown(): Promise<void> {
    // 按顺序关闭
    for (const client of this.clients) {
      await client.disconnect();
    }
    for (const edge of this.edges) {
      await edge.stop();
    }
    await this.hub.stop();
  }
  
  // 辅助方法
  getHub(): HubServer { return this.hub; }
  getEdge(index = 0): EdgeServer { return this.edges[index]; }
  getClient(index = 0): TestClient { return this.clients[index]; }
  
  // 等待条件
  async waitFor(
    condition: () => boolean,
    timeout = 5000,
    interval = 100
  ): Promise<void> {
    const start = Date.now();
    while (!condition()) {
      if (Date.now() - start > timeout) {
        throw new Error('Timeout waiting for condition');
      }
      await new Promise(r => setTimeout(r, interval));
    }
  }
}

// 使用
describe('Text Message Routing', () => {
  const env = new IntegrationTestEnv();
  
  beforeAll(() => env.setup({ edgeCount: 2 }));
  afterAll(() => env.teardown());
  
  it('routes text across edges', async () => {
    const client1 = env.getClient(0);
    const client2 = env.getClient(1);
    
    await client1.sendTextMessage('Hello');
    await env.waitFor(() => client2.receivedMessages.length > 0);
    
    expect(client2.receivedMessages[0]).toContain('Hello');
  });
});
```

### 4.2 性能测试框架

```typescript
// tests/helpers/perf-test.ts

export class PerfTestRunner {
  private results: Map<string, number[]> = new Map();
  
  async measure(name: string, fn: () => Promise<void>): Promise<number> {
    const start = performance.now();
    await fn();
    const duration = performance.now() - start;
    
    const existing = this.results.get(name) || [];
    existing.push(duration);
    this.results.set(name, existing);
    
    return duration;
  }
  
  async benchmark(
    name: string,
    fn: () => Promise<void>,
    iterations = 100
  ): Promise<{
    iterations: number;
    totalMs: number;
    avgMs: number;
    minMs: number;
    maxMs: number;
    p95Ms: number;
  }> {
    const times: number[] = [];
    
    for (let i = 0; i < iterations; i++) {
      const duration = await this.measure(name, fn);
      times.push(duration);
    }
    
    times.sort((a, b) => a - b);
    
    return {
      iterations,
      totalMs: times.reduce((a, b) => a + b, 0),
      avgMs: times.reduce((a, b) => a + b, 0) / times.length,
      minMs: times[0],
      maxMs: times[times.length - 1],
      p95Ms: times[Math.floor(times.length * 0.95)],
    };
  }
  
  report(): string {
    const lines: string[] = ['Performance Report', '='.repeat(40)];
    
    for (const [name, times] of this.results) {
      const sorted = times.sort((a, b) => a - b);
      const avg = sorted.reduce((a, b) => a + b, 0) / sorted.length;
      lines.push(`${name}:`);
      lines.push(`  Count: ${sorted.length}`);
      lines.push(`  Avg: ${avg.toFixed(2)}ms`);
      lines.push(`  Min: ${sorted[0].toFixed(2)}ms`);
      lines.push(`  Max: ${sorted[sorted.length - 1].toFixed(2)}ms`);
      lines.push(`  P95: ${sorted[Math.floor(sorted.length * 0.95)].toFixed(2)}ms`);
    }
    
    return lines.join('\n');
  }
}

// 使用

    expect(result.p95Ms).toBeLessThan(0.1); // P95 < 0.1ms
```
